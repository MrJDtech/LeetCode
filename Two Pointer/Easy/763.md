class Solution {
    public List<Integer> partitionLabels(String s) {

        int[] max_i= new int[26];
        ArrayList<Integer> AL= new ArrayList<>();
        int start=0;
        int end=0;
        for(int i=0;i<s.length();i++){
            max_i[s.charAt(i)-'a']=i;
        }

        for(int i=0;i<s.length();i++){
            end=Math.max(end,max_i[s.charAt(i)-'a']);
            if(i==end){
                AL.add(end-start+1);
                start= i+1;
            }
        }
        return AL;
    }
}
